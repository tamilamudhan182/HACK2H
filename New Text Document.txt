    <script>
        // Tab Switching
        function switchTab(tabName) {
            // Hide all tabs
            document.getElementById('routesTab').classList.add('hidden');
            document.getElementById('settingsTab').classList.add('hidden');
            document.getElementById('walletTab').classList.add('hidden');
            document.getElementById('historyTab').classList.add('hidden');

            // Show selected tab
            if (tabName === 'routes') {
                document.getElementById('routesTab').classList.remove('hidden');
            } else if (tabName === 'settings') {
                document.getElementById('settingsTab').classList.remove('hidden');
            } else if (tabName === 'wallet') {
                document.getElementById('walletTab').classList.remove('hidden');
            } else if (tabName === 'history') {
                document.getElementById('historyTab').classList.remove('hidden');
            }
        }

        // Bottom Navigation
        function switchBottomTab(element, tabName) {
            document.querySelectorAll('.nav-item').forEach(item => item.classList.remove('active'));
            element.classList.add('active');
            switchTab(tabName);
        }

        // Swap Locations
        function swapLocations() {
            const fromInput = document.getElementById('fromInput');
            const toInput = document.getElementById('toInput');
            const temp = fromInput.value;
            fromInput.value = toInput.value;
            toInput.value = temp;
        }

        // Toggle Switch
        function toggleSwitch(element) {
            element.classList.toggle('active');
        }

        // AI Assistant
        function openAIAssistant() {
            document.getElementById('modalOverlay').classList.add('active');
            document.getElementById('aiChatModal').classList.add('active');
        }

        // QR Wallet is now a tab explanation triggers
        function openQRWallet() {
            switchTab('wallet');
            switchBottomTab(document.querySelectorAll('.nav-item')[1], 'wallet');
        }

        // Close Modals
        function closeModals() {
            document.getElementById('modalOverlay').classList.remove('active');
            document.getElementById('aiChatModal').classList.remove('active');
            document.getElementById('qrWalletModal').classList.remove('active');
        }

        // Chat Functionality
        function sendMessage() {
            const input = document.getElementById('chatInput');
            const message = input.value.trim();

            if (message) {
                // Add user message
                addMessage(message, 'user');
                input.value = '';

                // Simulate AI response
                setTimeout(() => {
                    const responses = [
                        "I found 3 route options for you! The cheapest route costs ₹45 and takes 65 minutes using Metro + Bus. Would you like more details?",
                        "All stations on this route have wheelchair access with elevators and ramps. I can guide you to the accessible entrance!",
                        "I can translate the station announcements for you. Which language would you prefer?",
                        "The next metro arrives in 5 minutes. Platform 2, heading towards Central Station. Shall I set a reminder?",
                        "Your QR wallet has ₹850. This journey will cost ₹45. You'll have ₹805 remaining after this trip."
                    ];
                    const randomResponse = responses[Math.floor(Math.random() * responses.length)];
                    addMessage(randomResponse, 'ai');
                }, 1000);
            }
        }

        function handleChatEnter(event) {
            if (event.key === 'Enter') {
                sendMessage();
            }
        }

        function addMessage(text, type) {
            const messagesContainer = document.getElementById('chatMessages');
            const messageDiv = document.createElement('div');
            messageDiv.className = `message ${type}`;

            const avatar = document.createElement('div');
            avatar.className = 'message-avatar';
            avatar.textContent = type === 'ai' ? '🤖' : '👤';

            const content = document.createElement('div');
            content.className = 'message-content';
            content.textContent = text;

            messageDiv.appendChild(avatar);
            messageDiv.appendChild(content);
            messagesContainer.appendChild(messageDiv);

            // Scroll to bottom
            messagesContainer.scrollTop = messagesContainer.scrollHeight;
        }

        // Language buttons
        document.querySelectorAll('.lang-btn').forEach(btn => {
            btn.addEventListener('click', function () {
                document.querySelectorAll('.lang-btn').forEach(b => b.classList.remove('active'));
                this.classList.add('active');
            });
        });

        // Search Routes
        function searchRoutes() {
            document.getElementById('routeSearchState').style.display = 'none';

        // Show Route Details
        function showRouteDetails(type) {
            document.getElementById('routeResults').style.display = 'none';
            const details = document.getElementById('journeyDetails');
            details.classList.add('active');

            // Simple mock logic to update details based on type
            const badge = document.getElementById('detailBadge');
            const price = document.getElementById('detailPrice');

            if (type === 'cheapest') {
                badge.className = 'route-badge badge-cheapest';
                badge.innerText = '💰 Cheapest';
                price.innerText = '₹45';
            } else if (type === 'fastest') {
                badge.className = 'route-badge badge-fastest';
                badge.innerText = '⚡ Fastest';
                price.innerText = '₹280';
            } else if (type === 'accessible') {
                badge.className = 'route-badge badge-accessible';
                badge.innerText = '♿ Accessible';
                price.innerText = '₹165';
            }
        }
